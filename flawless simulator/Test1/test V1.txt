// HARDWARE PINS
int MOTOR_L_PWM = 9; int MOTOR_L_DIR = 7;
int MOTOR_R_PWM = 10; int MOTOR_R_DIR = 4;
int HORN_PIN = 8;

constexpr int DEFAULT_LINE_SKIP = 35;

// FIXED: Changed pins to A0, A1, A2, A3
constexpr int sensorsPins[NUM_SENSORS] = {A0, A1, A2, A3}; 
constexpr int sensorsRef[NUM_SENSORS] = {511, 511, 511, 511};

  trac_cross(100);
  fowardLineSkip();

void fd2(int Leftspeed, int RightSpeed){
  digitalWrite(MOTOR_L_DIR, LOW);
  digitalWrite(MOTOR_R_DIR, LOW);

  Leftspeed = constrain(Leftspeed, -100, 100);
  RightSpeed = constrain(RightSpeed, -100, 100);

  analogWrite(MOTOR_L_PWM, (Leftspeed * 255) / 100); 
  analogWrite(MOTOR_R_PWM, (RightSpeed * 255) / 100);
}
void tl(int speed) {
  fd2(speed, 0);
}
void tr(int speed) {
  fd2(0, speed);
}

void sl(int p) {
  int v = (p * 255) / 100;
  digitalWrite(MOTOR_L_DIR, HIGH);
  digitalWrite(MOTOR_R_DIR, LOW);
  analogWrite(MOTOR_L_PWM, v);
  analogWrite(MOTOR_R_PWM, v);
}
void sr(int p) {
  int v = (p * 255) / 100;
  digitalWrite(MOTOR_L_DIR, LOW);
  digitalWrite(MOTOR_R_DIR, HIGH);
  analogWrite(MOTOR_L_PWM, v);
  analogWrite(MOTOR_R_PWM, v);
}
void bk(int speed){
  digitalWrite(MOTOR_L_DIR, HIGH); digitalWrite(MOTOR_R_DIR, HIGH); // Reverse Polarity
  analogWrite(MOTOR_L_PWM, speed); 
  analogWrite(MOTOR_R_PWM, speed);
}

void readSensors(){
  // Loop through all 4 sensors
  for (int i = 0; i < NUM_SENSORS; i++) {
    // Read the analog value from the current sensor pin
    int val = analogRead(sensorsPins[i]);
    
    // Print the sensor number and its value
    Serial.print("S");
    Serial.print(i + 1); // Prints S1, S2, etc.
    Serial.print(":");
    Serial.print(val);
    Serial.print("\t");  // Adds a tab space between readings
  }
  
  Serial.println(); // Moves to the next line after printing all 4 sensors
  delay(100);       // Small delay to prevent crashing/spamming the Serial Monitor
}

void followUntilLine(int speed){
  while(1){
    if (analogRead(1) > sensorsRef[1] && analogRead(3) > sensorsRef[3]){
      break;
    }
    else {fd2(speed,speed);}
  }
  ao();
}
void followLine(int speed){
  // Check if both middle sensors (A1 and A2) detect the black line (value < reference)
  if (analogRead(A1) < sensorsRef[1] && analogRead(A2) < sensorsRef[2]) {
    fd2(speed,speed);
  }
  // If A1 (middle-left) is on white and A3 (right-most) meets the black line, turn right
  else if (analogRead(A1) > sensorsRef[1] && analogRead(A2) < sensorsRef[2]) {
    tr(50); 
  }                            
  // If A1 (middle-left) meets the black line and A3 (right-most) is on white, turn left
  else if (analogRead(A1) < sensorsRef[1] && analogRead(A2) > sensorsRef[2]) {
    tl(50); 
  }    
  else{
    fd2(speed,speed);
  }                        
}
void turnLeft(int speed){
  sl(100);
  delay(100);
  ao();
  while(1){
    if(analogRead(1) > sensorsRef[1]){
      break;
    }
    else sl(speed);

  }
  ao();
}
void turnRight(int speed){
  sr(100);
  delay(100);
  ao();
  delay(500);
  while(1){
    if(analogRead(2) > sensorsRef[2]){
      break;
    }
    else sr(speed);

  }
  ao();
}

void turnLeftLine(int speed){
  while(1){
    if(analogRead(1) > sensorsRef[1]){
      break;
    }
    else{tl(speed);}
  }
  ao();
}
void trac_CR(int speed){
  while(1){
    if(analogRead(3) > sensorsRef[3]){
      break;
    }
    else{followLine(speed);}
  }
  ao();
}
void trac_CL(int speed){
  while(1){
    if(analogRead(0) > sensorsRef[0]){
      break;
    }
    else{followLine(speed);}
  }
  ao();
}
void trac_cross(int speed){
  while(1){
    if(analogRead(0) > sensorsRef[0] && analogRead(3) > sensorsRef[3]){
      break;
    }
    else{followLine(speed);}
  }
  ao();
}
void fowardDelay(int speed, int delayMS){
  ao();
  fd2(speed, speed);
  delay(delayMS);
  ao();
}
void fowardLineSkip(){
  fd2(100,100);
  delay(DEFAULT_LINE_SKIP); // Replaced explicit delay value with constant
  ao();
}
void ao(){
  delay(100);
  analogWrite(MOTOR_L_PWM, 0);
  analogWrite(MOTOR_R_PWM, 0);
  delay(100);
}
void beep(){
  digitalWrite(HORN_PIN, HIGH); delay(500); digitalWrite(HORN_PIN, LOW);
}
